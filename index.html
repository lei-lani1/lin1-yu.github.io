解决Jupyter中文字体不显示问题

检查所有可用字体

import matplotlib.font_manager as fm
# 查看所有字体数量
all_fonts = [f.name for f in fm.fontManager.ttflist]
print(f"系统总字体数量: {len(all_fonts)}")
# 查看前20个字体名称
print("前20个字体:", all_fonts[:20])

检查macOS特定的中文字体

# macOS常见的中文字体名称
mac_fonts = [f.name for f in fm.fontManager.ttflist if any(keyword in f.name.lower() for keyword in 
 ['pingfang', 'hiragino', 'heiti', 'st', 'apple', 'ms', 'microsoft', 'noto', 'source han'])]

方案二: 使用plotly.express 或 plotly.graph_objects

#设置中文显示 
import plotly.express as px
import plotly.graph_objects as go

# 测试
import plotly.express as px 
import plotly.graph_objects as go 
import pandas as pd 

# 创建示例数据 
data = { 
 '日期': ['2026-01-01', '2026-01-02', '2026-01-03', '2026-01-04', '2026-01-05'], 
 '销售额': [100, 200, 150, 300, 250], 
 '成本': [60, 120, 90, 180, 150] 
} 
df = pd.DataFrame(data) 

# 使用plotly.express绘制折线图 
fig1 = px.line(df, x='日期', y='销售额', title='销售额趋势') 
fig1.show() 

# 使用plotly.graph_objects绘制柱状图 
fig2 = go.Figure() 
fig2.add_trace(go.Bar(x=df['日期'], y=df['销售额'], name='销售额')) 
fig2.add_trace(go.Bar(x=df['日期'], y=df['成本'], name='成本')) 
fig2.update_layout(title='销售额与成本对比', barmode='group') 
fig2.show()
